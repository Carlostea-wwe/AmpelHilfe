// RGB zu HSV Konvertierung function rgbToHsv(r, g, b) { r /= 255; g /= 255; b /= 255; const max = Math.max(r, g, b), min = Math.min(r, g, b); const d = max - min; let h = 0; const s = max === 0 ? 0 : d / max; const v = max; if (max !== min) { switch (max) { case r: h = (g - b) / d + (g < b ? 6 : 0); break; case g: h = (b - r) / d + 2; break; case b: h = (r - g) / d + 4; break; } h /= 6; } return [h * 360, s * 100, v * 100]; } // Analysiert das Canvas auf grüne LED-Pixel und liefert Bounding Box + Trefferquote function detectGreenLightRegion(ctx, width, height) { const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; let minX = width, maxX = 0, minY = height, maxY = 0; let greenPixelCount = 0; // Schrittweite 2 oder 4 für bessere Performance auf Mobilgeräten const step = 2; for (let y = 0; y < height; y += step) { for (let x = 0; x < width; x += step) { const i = (y * width + x) * 4; const r = data[i]; const g = data[i + 1]; const b = data[i + 2]; const [h, s, v] = rgbToHsv(r, g, b); // Filter für typisches Ampel-Grün (Smaragd/Türkis-Grün): // Hue: 140° - 175°, Sättigung > 40%, Helligkeit > 50% if (h >= 140 && h <= 175 && s >= 40 && v >= 50) { greenPixelCount++; if (x < minX) minX = x; if (x > maxX) maxX = x; if (y < minY) minY = y; if (y > maxY) maxY = y; } } } // Schwellenwert: Mindestens X grüne Pixel zusammenhängend gefunden? const hasGreenLight = greenPixelCount > 15; return { found: hasGreenLight, pixelCount: greenPixelCount, boundingBox: hasGreenLight ? { x: minX, y: minY, width: maxX - minX, height: maxY - minY } : null }; }
